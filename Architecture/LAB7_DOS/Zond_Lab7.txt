; Доказательная база по Lab7: протоколы автономных зондов (dosbox_autotest.ps1).
; Зонды НЕ являются пунктами задания: они фиксируют границы инструмента,
; найденные опытным путём до написания программ Lab7_1..Lab7_5.
;
; 1) TCC порождает ASM без .386 -> eax/cr0 в inline asm не собираются
;    (зонд P1: Undefined symbol: eax).
; 2) .386 в модуле ломает tlink /t -> требуется tlink /3 (зонды P4a/P4b).
; 3) CR0 читается машинным кодом под .286, но DOSBox отдаёт CR0=0000:0000,
;    поэтому бит NE из-emulyatora ne poluchit (зонд P4c, P5).
; 4) GLAVNOE: DOSBox 0.74-3 NE dostavlyaet nemaskirovannye iskuchenija x87.
;    Zond P5: CW=037B (maska bita 2 snyata), 1.0/0.0, FWAIT ->
;    SW=3904 (ZE=1), no schetchiki ISR(75h)=0 i ISR(16h)=0, ES(bit7)=0.
;    Vektor 75h do ustanovki = 0000:0000 (nichego v DOS tam net).
;    Vyvod: reagirovat na FLAG po FSTSW, a svoj obrabotchik vyzyvat
;    programmno (int 75h). Na realnom ZD dostavka avtomaticheskaja.
; 5) Baza FPU v TCC (zond P2): FINIT -> CW=037F, SW=0000;
;    zagolovok FSAVE = 14 bajt (386); 16.02f = 41 80 28 F6 (little-endian
;    F6 28 80 41); 0.0/0.0 -> SW=3901 (IE=1, TOP=7); ST(0) posle 1/0 =
;    7FFF 8000 0000 0000 0000 (x87 Infinity s javnym starshim bitom).
; 6) Pechat hex: cifry 30h..39h (oshibka 48h/57h daвала HHHH, zond P4c).
---------- _probe7\P1\probe.c ----------
/*==============================================================
  ZOND 1: bazovye Vozmoznosti FPU v DOSBox + TCC
  Ne chast zadanija - proverka, chto realno rabotaet:
    1) sobirajutsja li FPU-komandy v inline asm TCC
    2) kakoe znachenie bita NE v CR0 (opredeljaet vektor 75h ili 16h)
    3) kakoj razmer zagolovka daet FSAVE (94 ili 108 bajt)
    4) korrektno li fld/fadd/fstp dla float/double
    5) chto dajot 0.0/0.0 (NaN) i kakoj status slovo
  Vremennyj fail, udaljaetsja posle zamerov.
==============================================================*/
#include <stdio.h>
#include <dos.h>
#include <conio.h>

static unsigned char fbuf[120];     /* bufer FSAVE */
static unsigned      cw = 0, sw = 0, tw = 0;
static float         a = 16.02f, b = 3.5f, c = 0.0f, ssum = 0.0f;
static double        d1 = 16.02, d2 = 3.5, dsum = 0.0;
static unsigned long cr0v = 0;

/* ---------- pechat ---------- */
static void phex16(unsigned v) { printf("%04X", v); }
static void phex32(unsigned long v) { printf("%08lX", v); }

static void dumpb(unsigned char *p, int n, char *lbl)
{
    int i;
    printf("%s", lbl);
    for (i = 0; i < n; i++) printf("%02X ", p[i]);
    puts("");
}

void main(void)
{
    clrscr();
    puts("=== ZOND 1: FPU capabilities in DOSBox (TCC inline asm) ===");
    puts("");

    /* ---- 1. CR0 i bit NE ---- */
    asm mov  eax, cr0
    asm mov  cr0v, eax
    printf("CR0              = "); phex32(cr0v);
    printf("   bit NE (16) = "); printf("%d", (cr0v >> 16) & 1UL);
    puts("");
    puts("  NE=0 -> unmasked FPU exception goes to INT 75h (IRQ13)");
    puts("  NE=1 -> it goes to INT 16h");
    puts("");

    /* ---- 2. FINIT + FSTCW/FSTSW do nachala ---- */
    asm finit
    asm fstcw cw
    asm fstsw sw
    printf("after FINIT: CW = "); phex16(cw);
    printf("   SW = "); phex16(sw); puts("");

    /* ---- 3. fld/fadd/fstp dla float (3 chisla) ---- */
    asm finit
    asm fld  a
    asm fld  b
    asm fadd
    asm fld  c
    asm fadd
    asm fstp ssum
    printf("float  16.02 + 3.5 + 0.0 = "); printf("%.10f", ssum); puts("");
    printf("          expected exact  = 19.52"); puts("");

    /* ---- 4. to zhe dla double ---- */
    asm finit
    asm fld  d1
    asm fld  d2
    asm fadd
    asm fld  d2
    asm fadd
    asm fstp dsum
    printf("double 16.02 + 3.5 + 3.5 = "); printf("%.15lf", dsum); puts("");

    /* ---- 5. FSAVE: razmer zagolovka i soderzhimoe ST ---- */
    asm finit
    asm fld  a
    asm fld  b
    asm fld  c
    asm fsave fbuf
    dumpb(fbuf, 20, "FSAVE first 20 bytes: ");
    printf("  ST(0) raw (10 bytes) = "); dumpb(fbuf+14, 10, "");
    printf("  ST(1) raw (10 bytes) = "); dumpb(fbuf+24, 10, "");
    printf("  ST(2) raw (10 bytes) = "); dumpb(fbuf+34, 10, "");

    /* ---- 6. NaN: 0.0 / 0.0 ---- */
    asm finit
    asm fldz
    asm fldz
    asm fdiv
    asm fstsw sw
    asm fstp ssum
    printf("0.0/0.0 -> SW = "); phex16(sw); puts("");
    printf("  bit0 IE (invalid) = "); printf("%d", sw & 1); puts("");
    printf("  bit2 ZE (div0)    = "); printf("%d", (sw >> 2) & 1); puts("");
    printf("  bit7 ES (summary) = "); printf("%d", (sw >> 7) & 1); puts("");
    printf("  TOP (bits 13-11)  = "); printf("%d", (sw >> 11) & 7); puts("");

    /* ---- 7. peremennoe 16.02 kak IEEE754 single ---- */
    {
        unsigned long *pa = (unsigned long *) &a;
        printf("16.02f bits = "); phex32(*pa); puts("");
        printf("  sign/exp/mant: S=%d  E=%d  M=%06lX",
               (int)((*pa >> 31) & 1), (int)((*pa >> 23) & 0xFF), (unsigned long)(*pa & 0x7FFFFF));
        puts("");
    }
    puts("");
    puts("Press any key...");
#ifndef NOWAIT
    getch();
#endif
}

/* ZOND 2: bazovye FPU-komandy v TCC inline asm (bez 386-registrov)
   Vopros: kakie memoniki prinimaet TASM v pipeline TCC bez .386?
   Proveryaem: finit fld fadd fstp fstsw fstcw fldcw fsave fnclex fincstp fsqrt */
#include <stdio.h>
#include <dos.h>
#include <conio.h>

static unsigned char fbuf[128];
static unsigned cw=0, sw=0, cw2=0;
static float  f1=16.02f, f2=3.5f, f3=0.0f, fsum=0.0f;
static double d1=16.02, dsum=0.0;

static void pw(unsigned v){ printf("%04X", v); }
static void pd(unsigned v){ printf("%u", v); }
static void hd(unsigned char *p,int n){ int i; for(i=0;i<n;i++) printf("%02X ",p[i]); puts(""); }

void main(void)
{
    clrscr();
    puts("=== ZOND 2: FPU basics under TCC inline asm ===");

    asm finit
    asm fstcw cw
    asm fstsw  sw
    printf("FINIT -> CW="); pw(cw); printf("  SW="); pw(sw); puts("");

    asm finit
    asm fld f1
    asm fld f2
    asm fadd
    asm fld f3
    asm fadd
    asm fstp fsum
    printf("float  16.02+3.5+0.0 = "); printf("%.10f", fsum); puts("");

    asm finit
    asm fld d1
    asm fld d1
    asm fadd
    asm fstp dsum
    printf("double 16.02*2       = "); printf("%.15lf", dsum); puts("");

    asm finit
    asm fld f1
    asm fld f2
    asm fld f3
    asm fsave fbuf
    printf("FSAVE header(14): "); hd(fbuf,14);
    printf("  ST(0)= "); hd(fbuf+14,10);
    printf("  ST(1)= "); hd(fbuf+24,10);
    printf("  ST(2)= "); hd(fbuf+34,10);
    printf("  control= "); pw(fbuf[0]|(fbuf[1]<<8)); puts("");
    printf("  status = "); pw(fbuf[2]|(fbuf[3]<<8)); puts("");
    printf("  tag    = "); pw(fbuf[4]|(fbuf[5]<<8)); puts("");

    asm finit
    asm mov  ax, 003Fh
    asm mov  cw2, ax
    asm fldcw cw2
    asm fstcw cw
    printf("FLDCW 003Fh -> CW="); pw(cw); puts("");
    asm fnclex
    asm fincstp
    asm fld f1
    asm fsqrt
    asm fstp fsum
    printf("sqrt(16.02f) = "); printf("%.10f", fsum); puts("");

    asm finit
    asm fldz
    asm fldz
    asm fdiv
    asm fstsw sw
    asm fstp fsum
    printf("0.0/0.0 -> SW="); pw(sw);
    printf("  IE(bit0)="); pd(sw & 1);
    printf("  ZE(bit2)="); pd((sw>>2)&1);
    printf("  ES(bit7)="); pd((sw>>7)&1);
    printf("  TOP(13-11)="); pd((sw>>11)&7); puts("");

    asm finit
    asm fld f3
    asm fchs
    asm fsqrt
    asm fstsw sw
    asm fstp fsum
    printf("sqrt(-0.0) -> SW="); pw(sw); printf("  IE="); pd(sw&1); puts("");

    printf("16.02f  bytes: "); hd((unsigned char*)&f1,4);
    printf("16.02   bytes: "); hd((unsigned char*)&d1,8);
    puts("");
    puts("Press any key...");
#ifndef NOWAIT
    getch();
#endif
}
